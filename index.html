'متقن' : 'بحاجة لمراجعة'} `; }); } function toggleMastery(sIdx, lIdx) { if(!students[sIdx].mastery) students[sIdx].mastery = {}; students[sIdx].mastery[lIdx] = !students[sIdx].mastery[lIdx]; localStorage.setItem(getStorageKeyForStudents(), JSON.stringify(students)); renderMasteryTable(); showToast('✅ تم تحديث حالة الإتقان!'); }
